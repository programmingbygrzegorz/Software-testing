🎨 Edytor kolorów – bookmarklet

Narzędzie do zmiany kolorów znaczników HTML na dowolnej stronie internetowej. Działa bez instalacji – wystarczy przeciągnąć przycisk do paska zakładek.

Krok 1 – Przeciągnij przycisk do paska zakładek przeglądarki: Jeśli pasek zakładek nie jest widoczny, włącz go skrótem Ctrl + Shift + B

🎨 Edytor kolorów
Krok 2 – Wejdź na dowolną stronę (np. onet.pl) Kliknij zakładkę 🎨 Edytor kolorów w pasku przeglądarki. Panel pojawi się w prawym górnym rogu strony.
Krok 3 – Zmieniaj kolory Kliknij kwadracik koloru obok wybranego znacznika. Na Onecie używaj głównie DIV – tam jest większość treści.

Przycisk ↺ Przywróć oryginał cofa wszystkie zmiany.
Ponowne kliknięcie zakładki chowa panel.

📄 Kod źródłowy z komentarzami

Poniżej pełny kod skryptu z objaśnieniami jak działa każda część:

// ============================================================
// EDYTOR KOLORÓW – bookmarklet
// Skrypt wstrzykuje panel do dowolnej strony internetowej,
// który pozwala zmieniać kolory podstawowych znaczników HTML.
// ============================================================

(function () {

  // ----------------------------------------------------------
  // TOGGLE – jeśli panel już istnieje, usuń go i zakończ.
  // Dzięki temu ponowne kliknięcie zakładki chowa panel
  // zamiast tworzyć kolejny.
  // ----------------------------------------------------------
  const existing = document.getElementById('__colorPanel');
  if (existing) {
    existing.remove();
    return;
  }

  // ----------------------------------------------------------
  // LISTA ZNACZNIKÓW DO EDYCJI
  // Każdy obiekt zawiera:
  //   label    – nazwa wyświetlana w panelu
  //   selektor – selektor CSS używany przez querySelector
  // ----------------------------------------------------------
  const tagi = [
    { label: 'DIV',         selektor: 'div'  },
    { label: 'Nagłówki H1',  selektor: 'h1'   },
    { label: 'Nagłówki H2',  selektor: 'h2'   },
    { label: 'Nagłówki H3',  selektor: 'h3'   },
    { label: 'Akapity P',    selektor: 'p'    },
    { label: 'Linki A',      selektor: 'a'    },
    { label: 'Listy LI',     selektor: 'li'   },
    { label: 'Span',         selektor: 'span' },
  ];

  // ----------------------------------------------------------
  // TWORZENIE PANELU
  // Tworzymy nowy element div i ustawiamy mu style przez cssText.
  // position: fixed – panel przyklejony do rogu ekranu.
  // z-index: 2147483647 – maksymalna wartość, zawsze na wierzchu.
  // ----------------------------------------------------------
  const panel = document.createElement('div');
  panel.id = '__colorPanel';
  panel.style.cssText = `
    position: fixed;
    top: 20px;
    right: 20px;
    z-index: 2147483647;
    background: #fff;
    border: 1px solid #ddd;
    border-radius: 10px;
    padding: 16px;
    width: 260px;
    font-family: sans-serif;
    font-size: 14px;
    box-shadow: 0 4px 20px rgba(0,0,0,0.15);
  `;

  // Nagłówek panelu z tytułem i przyciskiem zamknięcia (✕)
  panel.innerHTML = `
    <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:12px;">
      <strong style="font-size:15px;">🎨 Edytor kolorów</strong>
      <button id="__closePanel" style="background:none;border:none;font-size:18px;cursor:pointer;color:#666;">✕</button>
    </div>
  `;

  // ----------------------------------------------------------
  // HELPER: rgbToHex()
  // Przeglądarka zwraca kolory w formacie: rgb(255, 0, 0)
  // Color picker wymaga formatu: #ff0000
  // Funkcja wyciąga liczby z rgb() i zamienia je na hex.
  // ----------------------------------------------------------
  function rgbToHex(rgb) {
    const result = rgb.match(/\d+/g);
    if (!result || result.length < 3) return '#000000';
    return '#' + result.slice(0, 3)
      .map(n => parseInt(n).toString(16).padStart(2, '0'))
      .join('');
  }

  // ----------------------------------------------------------
  // GENEROWANIE WIERSZY DLA KAŻDEGO ZNACZNIKA
  // forEach() iteruje po tablicy tagi[].
  // Dla każdego znacznika tworzy wiersz z etykietą i pickerem.
  // getComputedStyle() pobiera aktualny kolor elementu ze strony
  // i ustawia go jako wartość startową pickera.
  // addEventListener('input') reaguje na każdą zmianę koloru
  // i natychmiast aktualizuje wszystkie pasujące elementy.
  // ----------------------------------------------------------
  tagi.forEach(({ label, selektor }) => {

    const row = document.createElement('div');
    row.style.cssText = 'display:flex; justify-content:space-between; align-items:center; margin-bottom:8px;';

    const lbl = document.createElement('label');
    lbl.textContent = label;
    lbl.style.flex = '1';

    const input = document.createElement('input');
    input.type = 'color';
    input.style.cssText = 'width:40px; height:28px; border:none; cursor:pointer; border-radius:4px;';

    // Pobierz aktualny kolor pierwszego elementu na stronie
    const first = document.querySelector(selektor);
    if (first) {
      const computed = window.getComputedStyle(first).color;
      input.value = rgbToHex(computed);
    }

    // Zmiana koloru w pickerze → aktualizuj wszystkie elementy
    input.addEventListener('input', () => {
      document.querySelectorAll(selektor).forEach(el => {
        el.style.color = input.value;
      });
    });

    row.appendChild(lbl);
    row.appendChild(input);
    panel.appendChild(row);
  });

  // ----------------------------------------------------------
  // PRZYCISK RESET
  // Usuwa nadpisany kolor (style.color = '') ze wszystkich
  // elementów, co przywraca oryginalne style z CSS strony.
  // ----------------------------------------------------------
  const resetBtn = document.createElement('button');
  resetBtn.textContent = '↺ Przywróć oryginał';
  resetBtn.style.cssText = 'margin-top:10px;width:100%;padding:7px;background:#f44336;color:#fff;border:none;border-radius:6px;cursor:pointer;font-size:13px;';
  resetBtn.addEventListener('click', () => {
    tagi.forEach(({ selektor }) => {
      document.querySelectorAll(selektor).forEach(el => {
        el.style.color = '';
      });
    });
  });
  panel.appendChild(resetBtn);

  // ----------------------------------------------------------
  // WSTRZYKNIĘCIE PANELU DO STRONY
  // appendChild() dodaje panel jako ostatni element w body –
  // pojawia się na stronie bez przeładowania.
  // ----------------------------------------------------------
  document.body.appendChild(panel);

  // Obsługa przycisku zamknięcia ✕
  document.getElementById('__closePanel').addEventListener('click', () => {
    panel.remove();
  });

})();
// Całość opakowana w IIFE (Immediately Invoked Function Expression)
// – funkcja wywołuje się natychmiast po wczytaniu.
// Zmienne nie trafiają do globalnego scope strony
// i nie kolidują z jej własnym kodem JavaScript.

Zmiany są tymczasowe – po odświeżeniu strony wróci oryginał.
